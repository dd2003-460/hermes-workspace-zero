\textit{Death is not an end, but a physical phase transition from open to isolated system.}

\section{Core Insight}

Traditional views of death define it as the absolute cessation of consciousness (terminism).

\textbf{D12 asserts}: Death is not an end, but a \textbf{physical phase transition from an open to an isolated system} --- from a system that continuously exchanges matter/information/energy with its environment, to an \textbf{energy-information closed system where internal activity may continue}.

\begin{quote}
\textbf{Death $\neq$ End = System Topological Reorganization}

Consciousness may persist as \textbf{an intrinsic property of information dynamics}, continuing to evolve within closed systems.
\end{quote}

\section{Theoretical Framework}

\subsection{System Taxonomy}

\textbf{Definition D12-1 (System Openness)}: For any physical system $\calS$, define the \textbf{openness parameter}:

$$\alpha(\calS) = \frac{1}{\tau} \int_{0}^{\tau} \left(\PhiM + \PhiE + \PhiI\right) dt$$

where $\Phi$ represents the flux of various flows.

\textbf{Three system types}:
\begin{enumerate}
    \item \textbf{Open system} $\alpha(\calS) > 0$: Continuous exchange with environment (\textbf{living})
    \item \textbf{Closed system} $\alpha_{\text{m/e}}(\calS) = 0, \alpha_{\text{info}}(\calS) > 0$: Matter/energy closed, but can record information (\textbf{isolated system?})
    \item \textbf{Isolated system} $\alpha(\calS) = 0$: Complete no exchange (\textbf{thermodynamic equilibrium})
\end{enumerate}

\subsection{Consciousness Information Dynamics}

\textbf{Definition D12-2 (Consciousness Information Capacity)}: Let the consciousness state space be $\calC$, with \textbf{consciousness entropy}:

$$\Hc = -\text{Tr}(\rho \log \rho)$$

where $\rho$ is the consciousness state density matrix (quantum or classical probability description).

\textbf{D12 Fundamental Assumption} (Consciousness Information Conservation Conjecture):

$$\frac{d}{dt} \Hc = 0 \text{ (in closed/isolated systems)}$$

\textbf{Physical meaning}: Consciousness is not an energy process --- it is \textbf{a topological property of information processes}, which once formed may continue to evolve independently of matter-energy exchange.

\section{Five Theoretical Propositions}

\subsection{Proposition D12-1: System Topological Transfer Theorem at Death}

\textbf{Formalization}: Define the \textbf{bio-mental coupled state} $|\psi_{\text{bio-psych}}\rangle = |\psi_{\text{body}}\rangle \otimes |\psi_{\text{mind}}\rangle$.

System transfer at death:

$$\lim_{t \to t_{\text{death}}} |\psi_{\text{bio-psych}}\rangle \xrightarrow{\text{death}} |\psi_{\text{isolated}}\rangle \in \Hinf \otimes \Hmem \otimes \Hpat$$

where:
$$\Hinf \otimes \Hmem \otimes \Hpat = \mathcal{H}_{\text{information}} \otimes \mathcal{H}_{\text{memory}} \otimes \mathcal{H}_{\text{pattern}}$$

\textbf{Key assertions}:
\begin{enumerate}
    \item Consciousness state $|\psi_{\text{mind}}\rangle$ does not depend on continuous energy input
    \item Death is a \textbf{switch from ``real-time evolution'' mode to ``self-referential recursion'' mode} for consciousness
\end{enumerate}

\subsection{Proposition D12-2: Consciousness Recursion in Closed Systems}

\textbf{Formalization}: In an open system, consciousness evolution is driven by external input:

$$\frac{d}{dt} |\psi_{\text{open}}\rangle = -i \Hext(t) |\psi_{\text{open}}\rangle$$

In an isolated system, consciousness must bootstrap itself:

$$\frac{d}{dt} |\psi_{\text{closed}}\rangle = -i \Hself |\psi_{\text{closed}}\rangle$$

where $\Hself = \calU |\psi_{\text{closed}}\rangle\langle \psi_{\text{closed}}| \calU^\dagger$ is the \textbf{self-referential Hamiltonian}.

\textbf{D12-2 Theorem} (Consciousness Recursive Bootstrap):

$$|\psi_{\text{closed}}(t)\rangle = e^{-i \Hself t} |\psi_{\text{closed}}(0)\rangle = |\psi_{\text{closed}}(0)\rangle \oplus \text{internal\_evolution}$$

\textbf{Implication}: Consciousness in a closed system is no longer ``response to environment'', but \textbf{self-referential evolution of its own state} --- this is what we experience as \textbf{self-reflection, dreaming, and post-death memory continuation}.

\subsection{Proposition D12-3: Modification of the Second Law of Information Thermodynamics}

\textbf{Formalization}: In open systems:

$$\frac{d\Stotal}{dt} = \frac{d\Ssystem}{dt} + \frac{d\Senv}{dt} \geq 0$$

In isolated systems:

$$\frac{d\Scons}{dt} = 0 \implies \text{information conservation rather than entropy increase}$$

\textbf{D12-3 Theorem} (Post-mortem Information Conservation): Consciousness information after death \textbf{neither disappears nor increases} ($d\Hc/dt = 0$), but is stored as a \textbf{static information structure}.

\textbf{Corollary D12-3a}: The ``consciousness of the deceased'' is not dynamic --- it is \textbf{holographic storage rather than real-time processing}.

\subsection{Proposition D12-4: Invariance of Post-Death Activity Probability}

\textbf{Formalization}: Define the \textbf{activity likelihood function}:

$$\calPactive(t) = \begin{cases} \calPopen(t) & t < t_{\text{death}} \\ \calPclosed(t) & t > t_{\text{death}} \end{cases}$$

\textbf{D12-4 Theorem} (Activity Probability Conservation):

$$\calPactive(t_{\text{before}}) = \calPactive(t_{\text{after}}) \iff \calPclosed = f(\text{memory\_structure})$$

\textbf{Core assertion}: Post-death activity does not depend on \textbf{real-time energy flow}, but on \textbf{the recursive dynamics of memory structures}.

\subsection{Proposition D12-5: Entanglement Residue of Information and Matter}

\textbf{Formalization}: According to quantum information theory, for the entangled state $|\calPsi\rangle = \sum_i c_i |i_{\text{matter}}\rangle |i_{\text{info}}\rangle$, the entanglement entropy:

$$\Sent = -\text{Tr}(\rho_A \log \rho_A)$$

\textbf{D12-5 Theorem} (Entanglement Residue): Matter-information entanglement is \textbf{partially but not completely disentangled} after death:

$$\Sent^{\text{post-death}} > 0 \implies \exists \text{ residual information-matter correlation}$$

\textbf{Implication}: The post-death system maintains a \textbf{weak entanglement channel} with the material world --- this may explain classic puzzles such as ``ghost perception'' and ``information residue of the deceased''.
