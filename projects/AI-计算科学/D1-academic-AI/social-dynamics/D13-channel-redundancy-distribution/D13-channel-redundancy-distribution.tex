\section{Core Insight}

Classical information theory (Shannon) focuses on \textbf{point-to-point} reliable transmission. But the real world is a \textbf{one-to-many} system --- one information source broadcasting to receivers of unequal capacity.

\textbf{D13 asserts}: In a \textbf{capacity-asymmetric} one-to-many channel, fair information allocation requires \textbf{encoding redundancy} --- the source must add redundancy to ensure \textbf{information integrity for the weakest receiver}.

\begin{quote}
\textbf{Fairness = Minimum Receiver Capacity Guarantee}

\textbf{Redundancy} $R \geq 1/\beta_{\min}$, where $\beta_{\min} = \min(C_j)/C_{\text{source}}$
\end{quote}

\section{Theoretical Framework}

\subsection{Asymmetric One-to-Many Channel Model}

\textbf{Definition D13-1 (Asymmetric Channel System)}: Let source $S$ broadcast to receiver set $\calR = \{R_1, R_2, \ldots, R_N\}$, where each receiver's channel capacity is:

$$C_j = B_j \log_2\left(1 + \frac{P_s |h_j|^2}{\sigma_j^2}\right)$$

where:
\begin{itemize}
    \item $B_j$: receiver $j$'s \textbf{bandwidth} (pipe diameter)
    \item $P_s$: source transmission power
    \item $h_j$: channel gain
    \item $\sigma_j^2$: noise power
\end{itemize}

\textbf{Capacity asymmetry parameter}:

$$\betaj = \frac{C_j}{\calCmax} \in (0, 1]$$

$\betaj < 1$ indicates receiver $j$'s capability below the optimal receiver.

\subsection{Redundancy Allocation Function}

\textbf{Definition D13-2 (Redundancy Allocation Function)}: The source's \textbf{encoded codeword} length is $L_{\text{codeword}} = (1 + r) \cdot L_{\text{original}}$, where $r$ is the \textbf{redundancy level}.

\textbf{Fair allocation constraint}:

$$\forall j \in \calR: \quad C_j \geq \frac{L_{\text{codeword}}}{T}$$

where $T$ is the transmission time window.

\textbf{Minimum redundancy}:

$$r_{\min} = \max_{j} \left(\frac{L_j}{C_j \cdot T}\right) - 1$$

\section{Five Theoretical Propositions}

\subsection{Proposition D13-1: Fair Allocation Theorem (Big Pipe Theorem)}

\textbf{Formalization}: In a capacity-asymmetric channel system, the \textbf{fair transmission condition} is:

$$\calFfair \equiv \forall j_1, j_2 \in \calR: \quad \calInt(j_1) = \calInt(j_2) = \calOrig$$

where $\calInt(j)$ is the amount of information that receiver $j$ can \textbf{fully reconstruct}.

\textbf{D13-1 Theorem} (Big Pipe Theorem):

$$r_{\min} = \frac{1}{\min_j(\betaj)} - 1 = \frac{1 - \betamin}{\betamin}$$

\textbf{Physical meaning}: Redundancy is determined by the \textbf{weakest receiver} --- not maximizing average throughput, but \textbf{guaranteeing worst-case integrity}.

\subsection{Proposition D13-2: Nonlinear Cost of Redundancy}

\textbf{Formalization}: Define the \textbf{fairness cost function}:

$$\calDfair = \frac{r_{\min}}{1 + r_{\min}} \cdot \mathbb{E}_{j}[C_j]$$

\textbf{D13-2 Theorem} (Nonlinear Cost): As $\betamin \to 0$ (existence of extremely weak receivers):

$$\lim_{\betamin \to 0} \calDfair \to \infty$$

\textbf{Corollary D13-2a}: Guaranteeing information access for \textbf{extreme vulnerable groups} requires \textbf{nearly infinite redundancy}.

\textbf{Social implication}: In the AI domain, ``ensuring every reader can drink from the big pipe'' --- if we want everyone to achieve full understanding, the cost could be astronomical.

\subsection{Proposition D13-3: Hierarchical Redundancy Allocation Optimization}

\textbf{Formalization}: Divide receivers into $K$ \textbf{tiers} by capacity:

$$\calR = \bigcup_{k=1}^{K} \RcalK, \quad \betak \in [\betamink, \betamxk]$$

\textbf{D13-3 Theorem} (Optimal Hierarchical Redundancy):

$$\arg\min_{r^{(1)}, \ldots, r^{(K)}} \sum_{k=1}^{K} \frac{|\RcalK|}{|\calR|} \cdot \frac{r^{(k)}}{1 + r^{(k)}} \quad \text{s.t.} \quad \forall j \in \RcalK: C_j \geq r^{(k)}$$

\textbf{Optimal solution}:

$$r^{(k)*} = \frac{\betamxk - \betamink}{\betamink}$$

\textbf{Corollary}: Allocate different redundancy levels by capacity distribution --- \textbf{finer tiering yields higher overall efficiency}.

\subsection{Proposition D13-4: Redundancy--Efficiency--Fairness Triangle Trade-off}

\textbf{Formalization}: Define three optimization objectives:
\begin{itemize}
    \item \textbf{Efficiency}: $\calE = \frac{1}{N} \sum_j \frac{C_j}{\calCmax}$
    \item \textbf{Fairness}: $\calF = 1 - \frac{\max_j r_j - \min_j r_j}{\max_j r_j}$
    \item \textbf{Coverage}: $\calC = \frac{1}{N} \sum_j \mathbb{I}(\calInt(j) = \calOrig)$
\end{itemize}

\textbf{D13-4 Theorem} (Triangular Impossibility): No single redundancy allocation strategy \textbf{simultaneously maximizes} $\calE$, $\calF$, $\calC$:

$$\calE \uparrow \implies \calF \downarrow \lor \calC \downarrow$$

\textbf{Pareto frontier}:

$$\calPstar = \left\{(x, y, z) \mid x \cdot y \cdot z = \max\right\}$$

\textbf{Core insight}: Cannot optimize all dimensions with a single code --- requires \textbf{multi-code parallel transmission}.

\subsection{Proposition D13-5: Time-Slicing Fairness Theorem}

\textbf{Formalization}: If \textbf{spatial redundancy} cost is too high, time-slicing may be used instead:

$$T = \sum_{j=1}^{N} T_j, \quad T_j = \frac{L}{C_j}$$

\textbf{D13-5 Theorem} (Space-Time Duality): Time-slicing is equivalent to spatial redundancy under the following conditions:

$$\calFtime \equiv \calFred \iff \sum_{j} \frac{T}{T_j} \geq 1$$

\textbf{D13-5 Corollary}:
\begin{itemize}
    \item \textbf{High bandwidth / few receivers} $\rightarrow$ time-slicing is superior
    \item \textbf{Low bandwidth / many receivers} $\rightarrow$ spatial redundancy is superior
\end{itemize}

\section{Four Experimental Verification Protocols}

\subsection{Protocol D13-A: One-to-Many Fair Transmission Experiment}

\textbf{Verify D13-1}:
\begin{enumerate}
    \item Construct a simulated one-to-many channel (Python \texttt{simpy} or MATLAB)
    \item Test reception integrity under different redundancy levels
    \item \textbf{Expectation}: When $r = \frac{1-\betamin}{\betamin}$, all receivers can fully decode
\end{enumerate}

\subsection{Protocol D13-B: Redundancy--Efficiency--Fairness Pareto Frontier}

\textbf{Verify D13-4}:
\begin{enumerate}
    \item Scan redundancy parameters under fixed channel configuration
    \item Plot the 3D Pareto frontier
    \item \textbf{Expectation}: Three objectives cannot be simultaneously optimized; there exists a clear trade-off surface
\end{enumerate}

\subsection{Protocol D13-C: Hierarchical Redundancy Comparison Experiment}

\textbf{Verify D13-3}:
\begin{enumerate}
    \item Divide receivers into $K$ tiers by capacity
    \item Compare total efficiency of single redundancy vs. hierarchical redundancy
    \item \textbf{Expectation}: Hierarchical redundancy yields $> 15\\%$ efficiency gain when $K \geq 3$
\end{enumerate}

\subsection{Protocol D13-D: Space-Time Duality Verification}

\textbf{Verify D13-5}:
\begin{enumerate}
    \item Apply time-slicing and spatial redundancy to the same information set
    \item Measure fairness and efficiency under both strategies
    \item \textbf{Expectation}: When $T \cdot \sum_j(1/C_j) < T_{\text{threshold}}$, space-time strategies are equivalent
\end{enumerate}

\section{Theoretical Boundaries}

\subsection{Key Falsifiable Claims}
\begin{enumerate}
    \item \textbf{D13-1 Falsification}: No minimum redundancy determined by the weakest receiver exists
    \item \textbf{D13-2 Falsification}: Marginal cost of guaranteeing extreme vulnerability does not tend to infinity
    \item \textbf{D13-4 Falsification}: A single encoding strategy simultaneously optimizing efficiency/fairness/coverage exists
    \item \textbf{D13-5 Falsification}: The space-time duality condition does not hold
\end{enumerate}

\subsection{Theoretical Limitations}
\begin{enumerate}
    \item Does not consider \textbf{dynamic channel changes} (non-stationary scenarios)
    \item Does not model \textbf{receiver collaboration} (e.g., information relay)
    \item \textbf{Application-specific}: requires parameter adjustment for specific AI/communication scenarios
\end{enumerate}

\section{Final Theorem Statement}

\begin{tcolorbox}[colback=blue!5!white,colframe=blue!75!black,title=D13 Big Pipe Theorem]
\textbf{Given} information source $S$ broadcasting to $N$ receivers, with receiver capacity $C_j$ non-uniformly distributed, $\betaj = C_j/C_{\max}$.

\textbf{Then}: The minimum redundancy for fair transmission is:

$$\boxed{r_{\min} = \frac{1 - \betamin}{\betamin}}$$

\textbf{And}:
\begin{itemize}
    \item \textbf{Fairness cost} $\calDfair \to \infty$ when $\betamin \to 0$
    \item \textbf{Space-time duality} $\calFtime \equiv \calFred \iff \sum_j T/T_j \geq 1$
\end{itemize}

\textbf{That is}: In a capacity-asymmetric one-to-many system, fair information allocation requires \textbf{redundancy determined by the weakest receiver}, and guaranteeing extreme vulnerability requires \textbf{astronomical redundancy} --- this explains why ``AI serving everyone'' is so difficult.
\end{tcolorbox}
